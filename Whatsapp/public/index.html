1. Send Receipt with Optional Base64 PDF Attachment & Parallel Routing:

DECLARE
    l_body     CLOB;
    l_response CLOB;
BEGIN
    -- Construct receipt payload with optional Base64 PDF attachment
    l_body := '{'
        || '"phones": "'       || :P10_CUSTOMER_PHONE || '",' -- single or comma-separated
        || '"customerName": "' || :P10_CUSTOMER_NAME  || '",'
        || '"orderNumber": "'  || :P10_ORDER_ID       || '",'
        || '"totalAmount": "'  || :P10_TOTAL_AMOUNT   || '",'
        || '"senderId": "auto",'                              -- "auto" for round-robin parallel
        || '"delaySeconds": 6,'                               -- custom delay (6s)
        -- Optional PDF Invoice Base64 attachment:
        || '"fileName": "Invoice_' || :P10_ORDER_ID || '.pdf",'
        || '"mimeType": "application/pdf",'
        || '"fileBase64": "'   || :P10_PDF_BASE64     || '"'  -- pass NULL or base64
        || '}';

    apex_web_service.g_request_headers.DELETE;
    apex_web_service.g_request_headers(1).name  := 'Content-Type';
    apex_web_service.g_request_headers(1).value := 'application/json';
    apex_web_service.g_request_headers(2).name  := 'x-api-key';
    apex_web_service.g_request_headers(2).value := 'your-secret-api-key-change-this-12345';

    l_response := apex_web_service.make_rest_request(
        p_url         => 'http://localhost:3000/api/send-receipt',
        p_http_method => 'POST',
        p_body        => l_body
    );
END;

3. Send ONLY File or Image (Base64 PDF, Picture, Slip) in Oracle APEX:

DECLARE
    l_body     CLOB;
    l_response CLOB;
BEGIN
    -- Send ONLY a PDF document or Image with optional caption (No receipt template required!)
    l_body := '{'
        || '"phones": "'       || :P10_CUSTOMER_PHONE || '",' -- single or bulk (comma-separated)
        || '"senderId": "auto",'                              -- "auto" distributes across lines in parallel
        || '"delaySeconds": 6,'
        || '"fileName": "Invoice_' || :P10_ORDER_ID || '.pdf",'
        || '"mimeType": "application/pdf",'                   -- application/pdf, image/jpeg, image/png, etc.
        || '"fileBase64": "'   || :P10_FILE_BASE64    || '",'
        || '"caption": "Dear customer, please find your invoice attached."' -- optional caption text
        || '}';

    apex_web_service.g_request_headers.DELETE;
    apex_web_service.g_request_headers(1).name  := 'Content-Type';
    apex_web_service.g_request_headers(1).value := 'application/json';
    apex_web_service.g_request_headers(2).name  := 'x-api-key';
    apex_web_service.g_request_headers(2).value := 'your-secret-api-key-change-this-12345';

    l_response := apex_web_service.make_rest_request(
        p_url         => 'http://localhost:3000/api/send-file',
        p_http_method => 'POST',
        p_body        => l_body
    );
END;
